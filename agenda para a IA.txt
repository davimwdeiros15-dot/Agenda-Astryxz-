    async function toggleMicHardware() {
        const btn = document.getElementById('btn-voice-mic'), label = document.getElementById('voice-status-label'), selfCard = document.getElementById('voice-card-self');
        if (!micAtivo) {
            try {
                localAudioStream = await navigator.mediaDevices.getUserMedia({ audio: true });
                micAtivo = true; btn.classList.add('active-green'); label.textContent = "🎤 Ligado";
                audioContext = new (window.AudioContext || window.webkitAudioContext)();
                analyser = audioContext.createAnalyser(); microphoneSource = audioContext.createMediaStreamSource(localAudioStream);
                javascriptNode = audioContext.createScriptProcessor(2048, 1, 1);
                analyser.smoothingTimeConstant = 0.8; analyser.fftSize = 1024;
                microphoneSource.connect(analyser); analyser.connect(javascriptNode); javascriptNode.connect(audioContext.destination);
                javascriptNode.onaudioprocess = () => {
                    const array = new Uint8Array(analyser.frequencyBinCount); analyser.getByteFrequencyData(array);
                    let values = 0; for (let i = 0; i < array.length; i++) { values += array[i]; }
                    if ((values / array.length) > 12) { selfCard.classList.add('speaking'); } else { selfCard.classList.remove('speaking'); }
                };
            } catch (err) { alert("Microfone indisponível."); }
        } else {
            if(localAudioStream) localAudioStream.getTracks().forEach(track => track.stop());
            if(javascriptNode) javascriptNode.disconnect();
            micAtivo = false; btn.classList.remove('active-green'); label.textContent = "🎤 Desligado"; selfCard.classList.remove('speaking');
        }
    }

    /* BACKGROUND ANIMATION ENGINE (REESTRUTURADO PARA OS NOVOS TEMAS) */
    window.addEventListener('resize', setupCanvas);
    function setupCanvas() {
        canvas.width = window.innerWidth; 
        canvas.height = window.innerHeight; 
        particles = [];
        
        let pCount = currentTheme === 'aurora' ? 50 : 80;
        for(let i=0; i < pCount; i++) {
            particles.push({ 
                x: Math.random() * canvas.width, 
                y: Math.random() * canvas.height, 
                size: Math.random() * 4 + 1, 
                vx: (Math.random() - 0.5) * 0.8, 
                vy: currentTheme === 'sakura' || currentTheme === 'floresta' ? (Math.random() * 0.6 + 0.4) : (Math.random() - 0.5) * 0.6,
                angle: Math.random() * Math.PI * 2,
                weight: Math.random() * 0.5 + 0.2
            });
        }
    }

    function draw() {
        const bgFills = { 
            'aurora': '2, 6, 23', 
            'cyberpunk': '10, 10, 0', 
            'sakura': '42, 8, 19', 
            'oceano': '0, 11, 24', 
            'floresta': '2, 15, 4',
            'nuke': '8, 13, 2'
        };
        ctx.fillStyle = `rgba(${bgFills[currentTheme] || bgFills['aurora']}, 0.25)`; 
        ctx.fillRect(0, 0, canvas.width, canvas.height);

        // 1. DESENHAR FUNDOS COMPLEXOS ESTÁTICOS/ILUSTRAÇÕES VIA CÓDIGO ANTES DAS PARTÍCULAS
        if (currentTheme === 'floresta') {
            // Desenhar silhuetas de Árvores ao fundo do Campo
            ctx.fillStyle = "rgba(4, 26, 8, 0.4)";
            for(let i = 0; i < canvas.width; i += 120) {
                ctx.beginPath();
                ctx.moveTo(i, canvas.height);
                ctx.lineTo(i + 40, canvas.height - 180);
                ctx.lineTo(i + 80, canvas.height);
                ctx.fill();
                // Troncos
                ctx.fillStyle = "rgba(2, 12, 4, 0.5)";
                ctx.fillRect(i + 36, canvas.height - 60, 8, 60);
                ctx.fillStyle = "rgba(4, 26, 8, 0.4)";
            }
        } 
        else if (currentTheme === 'sakura') {
            // Desenhar Silhueta de um Templo Chinês/Japonês ao Fundo
            ctx.fillStyle = "rgba(24, 4, 10, 0.5)";
            let cx = canvas.width / 2 - 150;
            let cy = canvas.height - 250;
            // Base do templo
            ctx.fillRect(cx + 30, cy + 120, 240, 130);
            // Telhado inferior
            ctx.beginPath();
            ctx.moveTo(cx, cy + 120);
            ctx.quadraticCurveTo(cx + 150, cy + 100, cx + 300, cy + 120);
            ctx.lineTo(cx + 260, cy + 80);
            ctx.lineTo(cx + 40, cy + 80);
            ctx.closePath(); ctx.fill();
            // Andar de cima
            ctx.fillRect(cx + 60, cy + 30, 180, 50);
            // Telhado Superior do Templo
            ctx.beginPath();
            ctx.moveTo(cx + 20, cy + 30);
            ctx.quadraticCurveTo(cx + 150, cy, cx + 280, cy + 30);
            ctx.lineTo(cx + 220, cy - 20);
            ctx.lineTo(cx + 80, cy - 20);
            ctx.closePath(); ctx.fill();
        } 
        else if (currentTheme === 'cyberpunk') {
            // Desenhar Silhueta Abstrata de um Dragão Oriental Amarelo no Fundo de forma Vetorial
            ctx.strokeStyle = "rgba(255, 255, 0, 0.05)";
            ctx.lineWidth = 12;
            ctx.beginPath();
            // Corpo serpentinado cobrindo a tela
            ctx.moveTo(canvas.width * 0.1, canvas.height * 0.8);
            ctx.bezierCurveTo(canvas.width * 0.3, canvas.height * 0.2, canvas.width * 0.5, canvas.height * 0.9, canvas.width * 0.7, canvas.height * 0.3);
            ctx.quadraticCurveTo(canvas.width * 0.85, canvas.height * 0.1, canvas.width * 0.9, canvas.height * 0.4);
            ctx.stroke();
            
            // Cabeça do Dragão e Detalhes
            ctx.fillStyle = "rgba(255, 255, 0, 0.04)";
            ctx.beginPath();
            ctx.arc(canvas.width * 0.9, canvas.height * 0.4, 50, 0, Math.PI * 2);
            ctx.fill();

            // Raios Amarelos Aleatórios (Efeito tempestade elétrica Cyber)
            if (Math.random() < 0.15) {
                ctx.strokeStyle = "rgba(255, 255, 0, 0.4)";
                ctx.lineWidth = Math.random() * 3 + 1;
                ctx.beginPath();
                let rx = Math.random() * canvas.width;
                ctx.moveTo(rx, 0);
                ctx.lineTo(rx + (Math.random() - 0.5) * 100, canvas.height * 0.3);
                ctx.lineTo(rx + (Math.random() - 0.5) * 200, canvas.height * 0.7);
                ctx.lineTo(rx + (Math.random() - 0.5) * 50, canvas.height);
                ctx.stroke();
            }
        }
        else if (currentTheme === 'oceano') {
            // Ondas Orgânicas Fluidas de fundo
            waveOffset += 0.015;
            ctx.fillStyle = "rgba(0, 32, 64, 0.3)";
            ctx.beginPath();
            ctx.moveTo(0, canvas.height);
            for (let x = 0; x <= canvas.width; x += 20) {
                let y = canvas.height - 120 + Math.sin(x * 0.004 + waveOffset) * 25 + Math.cos(x * 0.008 + waveOffset) * 10;
                ctx.lineTo(x, y);
            }
            ctx.lineTo(canvas.width, canvas.height);
            ctx.closePath(); ctx.fill();
        }
        else if (currentTheme === 'nuke') {
            // Desenhar Logo Central de Radiação Tóxica Verde
            let midX = canvas.width / 2;
            let midY = canvas.height / 2;
            ctx.strokeStyle = "rgba(57, 255, 20, 0.04)";
            ctx.fillStyle = "rgba(57, 255, 20, 0.03)";
            
            // Círculo externo
            ctx.lineWidth = 6;
            ctx.beginPath(); ctx.arc(midX, midY, 140, 0, Math.PI*2); ctx.stroke();
            
            // Abas de hélice da radiação
            for(let a=0; a<3; a++) {
                let startAng = (a * 120 * Math.PI / 180) + (Date.now()/5000);
                let endAng = startAng + (60 * Math.PI / 180);
                ctx.beginPath();
                ctx.moveTo(midX, midY);
                ctx.arc(midX, midY, 120, startAng, endAng);
                ctx.closePath(); ctx.fill();
            }
            // Centro protetor
            ctx.fillStyle = "rgba(8, 13, 2, 1)";
            ctx.beginPath(); ctx.arc(midX, midY, 35, 0, Math.PI*2); ctx.fill();
            ctx.fillStyle = "rgba(57, 255, 20, 0.04)";
            ctx.beginPath(); ctx.arc(midX, midY, 25, 0, Math.PI*2); ctx.fill();

            // Bombas de Esgoto Líquido / Tonéis nas laterais inferiores
            ctx.fillStyle = "rgba(25, 45, 10, 0.5)";
            ctx.fillRect(40, canvas.height - 140, 70, 140); // Bomba Esquerda
            ctx.fillRect(canvas.width - 110, canvas.height - 140, 70, 140); // Bomba Direita
            ctx.fillStyle = "rgba(57, 255, 20, 0.3)";
            ctx.fillRect(45, canvas.height - 130, 60, 10); // Detalhe verde vazando líquido
            ctx.fillRect(canvas.width - 105, canvas.height - 130, 60, 10);
        }

        // 2. RENDERIZAR E MOVER AS PARTÍCULAS DE ACORDO COM O TEMA ESCOLHIDO
        particles.forEach(p => {
            p.x += p.vx; 
            p.y += p.vy;

            // Comportamentos e estilizações customizadas
            if (currentTheme === 'sakura') {
                // Pétalas de Sakura caem de lado
                p.vx = 0.5 + Math.sin(p.angle) * 0.2;
                ctx.fillStyle = `rgba(244, 63, 94, ${p.weight})`;
                ctx.beginPath();
                ctx.ellipse(p.x, p.y, p.size * 1.5, p.size, Math.PI / 4, 0, Math.PI * 2);
                ctx.fill();
            } 
            else if (currentTheme === 'floresta') {
                // Folhas verdes caindo no campo
                p.vx = Math.sin(p.angle) * 0.3;
                ctx.fillStyle = `rgba(16, 185, 129, ${p.weight + 0.2})`;
                ctx.beginPath();
                ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                ctx.fill();
            } 
            else if (currentTheme === 'oceano') {
                // Bolhas subindo de baixo para cima
                p.vy = -Math.abs(p.vy) - 0.2;
                ctx.strokeStyle = "rgba(56, 189, 248, 0.4)";
                ctx.lineWidth = 1;
                ctx.beginPath();
                ctx.arc(p.x, p.y, p.size + 1, 0, Math.PI * 2);
                ctx.stroke();
            } 
            else if (currentTheme === 'cyberpunk') {
                // Poeira binária amarela brilhante
                ctx.fillStyle = "rgba(255, 255, 0, 0.7)";
                ctx.fillRect(p.x, p.y, p.size, p.size);
            }
            else if (currentTheme === 'nuke') {
                // Fumaça/Gás radioativo verde subindo das bombas
                p.vy = -Math.abs(p.vy) - 0.4;
                p.vx += (Math.random() - 0.5) * 0.3;
                ctx.fillStyle = `rgba(57, 255, 20, ${p.weight * 0.4})`;
                ctx.beginPath();
                ctx.arc(p.x, p.y, p.size * 3, 0, Math.PI * 2);
                ctx.fill();
            }
            else {
                // Aurora / Padrão Estelar comum
                ctx.beginPath(); 
                ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2); 
                ctx.fillStyle = 'rgba(255,255,255,0.15)'; 
                ctx.fill();
            }

            // Loop das partículas pelas bordas
            if(p.x < 0) p.x = canvas.width; 
            if(p.x > canvas.width) p.x = 0;
            if(p.y < 0) p.y = canvas.height; 
            if(p.y > canvas.height) p.y = 0;
            p.angle += 0.01;
        });

        requestAnimationFrame(draw);
    }
    setupCanvas(); 
    draw();
